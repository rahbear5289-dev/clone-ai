/**
 * Autonomous web scraper fallback via DuckDuckGo HTML search
 */
async function searchDuckDuckGo(query: string): Promise<{ url: string; content: string }[]> {
  try {
    const resp = await axios.get(
      `https://html.duckduckgo.com/html/?q=${encodeURIComponent(query)}`,
      {
        headers: {
          'User-Agent':
            'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/122.0.0.0 Safari/537.36',
          Accept: 'text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8'
        },
        timeout: 8000
      }
    )
    const $ = load(resp.data)
    const results: { url: string; content: string }[] = []
    $('.result').each((i, el) => {
      if (i >= 5) return
      const title = $(el).find('.result__title').text().trim()
      const snippet = $(el).find('.result__snippet').text().trim()
      const rawUrl = $(el).find('.result__url').attr('href') || ''
      if (title && snippet) {
        results.push({
          url: rawUrl || title,
          content: `${title}: ${snippet}`
        })
      }
    })
    return results
  } catch {
    return []
  }
}

export default function registerDeepResearch({ ipcMain }: { ipcMain: IpcMain }) {

  ipcMain.handle(

    async (event, { query, tavilyKey, groqKey, geminiKey }) => {
      try {
        if (!query || !query.trim()) {
          return { success: false, error: 'Empty query provided for research.' }
        }

        const emitProgress = (status: string, file: string, totalFound: number) => {
          try {
            event.sender.send('deep-research-progress', { status, file, totalFound })
          } catch {}
        }

        emitProgress('scanning', 'Scanning global intelligence and web sources...', 1)

        let searchResults: { url: string; content: string }[] = []

        // 1. Try Tavily Search if API key is provided
        if (tavilyKey && tavilyKey.trim()) {
          try {
            emitProgress('scanning', 'LAINEX and Tavily Neural Search Active...', 1)
            const tvly = tavily({ apiKey: tavilyKey.trim() })
            const tavilyData = await tvly.search(query, {
              searchDepth: 'advanced',
              includeAnswer: true,
              maxResults: 5
            })
            if (tavilyData?.results && Array.isArray(tavilyData.results)) {
              searchResults = tavilyData.results.map((r: any) => ({
                url: r.url || '',
                content: r.content || ''
              }))
            }
          } catch {
            // Silently fallback if Tavily fails or is unreachable
          }
        }

        // 2. Fallback to Autonomous DuckDuckGo search if Tavily was unavailable or failed
        if (searchResults.length === 0) {
          try {
            emitProgress('scanning', 'Autonomous Web Crawler Active...', 1)
            searchResults = await searchDuckDuckGo(query)
          } catch {}

                }

                }
              } catch {}
            }
          } catch {}
        }

        // 5. If still no summary, use extracted web snippets directly
        if (!summary) {
          if (searchResults.length > 0) {
            summary = searchResults
              .map((r) => `• **${r.url}**: ${r.content.slice(0, 300)}...`)
              .join('\n\n')
          } else {
            throw new Error('Research failed: Please configure Gemini or Groq API key in Settings.')
          }
        }

        emitProgress('embedded', 'Research synthesis complete...', 3)

        return { success: true, summary }
      } catch (error) {
        return { success: false, error: String(error) }
      }
    }
